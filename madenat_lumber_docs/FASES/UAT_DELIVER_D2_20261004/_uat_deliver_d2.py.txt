def counts():
    d = {}
    for m in MODELS:
        try:
            d[m] = E[m].search_count([])
        except Exception as e:  # noqa: BLE001
            d[m] = 'ERR:%s' % e
    return d


def delta(a, b):
    return {k: (a[k] - b[k])
            if isinstance(a.get(k), int) and isinstance(b.get(k), int)
            else '?' for k in MODELS}


def nophys(d):
    return all(d.get(m, 1) == 0 for m in NOPHYS)


A = E['madenat.audit.log']


def audits(sid, atype):
    return A.search([('action_type', '=', atype), ('batch_id', '=', str(sid))])


M = E['mail.message']


def chatter(sid, needle):
    return M.search([('model', '=', 'lumber.export.shipment'),
                     ('res_id', '=', sid), ('body', 'like', needle)])


def attempt(label, fn):
    try:
        r = fn()
        tag = r.get('tag') if isinstance(r, dict) else ''
        P(label, 'NO_ERROR', tag)
        return r
    except Exception as e:  # noqa: BLE001
        P(label, type(e).__name__, str(e).replace('\n', ' | ')[:200])
        return None


def hash_ok(a):
    return hashlib.sha256((a.audit_snapshot or '').encode('utf-8')).hexdigest() == a.audit_hash


def protected():
    def nm(model, rid):
        r = E[model].browse(rid)
        return r.name if r.exists() else 'MISSING'

    def st(model, rid):
        r = E[model].browse(rid)
        if not r.exists():
            return 'MISSING'
        return getattr(r, 'state', nm(model, rid))

    pr = {}
    for i in (1537, 1528, 1885, 2128, 1867, 1868, 1644, 1645, 1646, 1704, 1705, 1706, 1707):
        pr['ship_%d' % i] = (nm('lumber.export.shipment', i), st('lumber.export.shipment', i))
    for i in (1509, 1500, 1629, 1630, 1631, 1678, 1679, 1680):
        pr['cont_%d' % i] = (nm('lumber.container', i), st('lumber.container', i))
    for i in (619, 620):
        pr['vessel_%d' % i] = nm('shipping.vessel', i)
    return pr

def approve_docs(ship):
    for d in ship.document_ids:
        d.file_data = base64.b64encode(('uat-%s' % d.id).encode('utf-8')).decode('utf-8')
    ship.invalidate_recordset()


def mk_container(name, ship, state, line_state=None):
    c = E['lumber.container'].create({
        'name': name, 'container_type': '40HC', 'shipment_id': ship.id, 'state': state})
    if line_state:
        lot = E['stock.lot'].create({'name': 'UATDLVLOT-%s' % name,
                                     'product_id': PROD.id, 'company_id': env.company.id})
        E['lumber.shipment.line'].create({
            'container_id': c.id, 'shipment_id': ship.id,
            'lot_id': lot.id, 'rebaja_state': line_state})
    c.invalidate_recordset()
    return c


def mk_ship(name):
    return E['lumber.export.shipment'].create({
        'name': PREF + name, 'customer_id': partner.id,
        'booking_reference': PREF + 'B-' + name, 'vessel_id': vessel.id})
