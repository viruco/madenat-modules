import json, hashlib, datetime

def P(*a):
    print('UAT|', *a)

def sect(t):
    P(''); P('======== ' + t + ' ========')

MODELS = ['stock.move', 'stock.move.line', 'stock.picking', 'stock.quant',
          'stock.lot', 'lumber.export.shipment', 'lumber.container',
          'lumber.shipment.line', 'shipping.vessel', 'shipping.voyage',
          'madenat.audit.log', 'madenat.sv.operation', 'madenat.sv.imputation',
          'stock.lot.cost.line', 'lumber.shipment.cost.line',
          'lumber.billing.consolidation', 'account.move', 'account.payment']

def counts():
    d = {}
    for m in MODELS:
        try:
            d[m] = env[m].search_count([])
        except Exception as e:
            d[m] = 'ERR:%s' % e
    return d

def delta(a, b):
    return {k: (a[k] - b[k]) if isinstance(a.get(k), int) and isinstance(b.get(k), int)
            else '?' for k in MODELS}

def st(model, rid):
    r = env[model].browse(rid)
    return r.state if r.exists() and 'state' in r._fields else (
        r.name if r.exists() else 'MISSING')

def nm(model, rid):
    r = env[model].browse(rid)
    return r.name if r.exists() else 'MISSING'

def lot_split(lot_id):
    lot = env['stock.lot'].browse(lot_id)
    if not lot.exists():
        return 'MISSING'
    qs = env['stock.quant'].search([('lot_id', '=', lot_id)])
    internal = round(sum(q.quantity for q in qs if q.location_id.usage == 'internal'), 4)
    customer = round(sum(q.quantity for q in qs if q.location_id.usage == 'customer'), 4)
    return {'name': lot.name, 'internal': internal, 'customer': customer}

def return_audit(sid):
    return env['madenat.audit.log'].search([
        ('action_type', '=', 'shipment_return_to_draft'),
        ('batch_id', '=', str(sid))])

def success_chatter(sid):
    return env['mail.message'].search([
        ('model', '=', 'lumber.export.shipment'), ('res_id', '=', sid),
        ('body', 'like', 'Retorno seguro a borrador')])

def mk_cont(name, shipment):
