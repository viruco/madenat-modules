MODELS = ['stock.move', 'stock.move.line', 'stock.picking', 'stock.quant',
          'stock.lot', 'stock.lot.cost.line', 'lumber.shipment.cost.line',

def counts():
    d = {}
    for m in MODELS:
        try:
            d[m] = E[m].search_count([])
        except Exception as e:  # noqa: BLE001
            d[m] = 'ERR:%s' % e
    return d


def delta(a, b):
    return {k: (a[k] - b[k]) if isinstance(a.get(k), int) and isinstance(b.get(k), int)
            else '?' for k in MODELS}


def b01_audits(sid):
    return E['madenat.audit.log'].search([
        ('action_type', '=', 'b01_departure_exception'), ('batch_id', '=', str(sid))])


def dev_audits(sid):
    return E['madenat.audit.log'].search([
        ('action_type', '=', 'deviation'), ('batch_id', '=', str(sid))])


def chatter(sid, needle):
    return E['mail.message'].search([
        ('model', '=', 'lumber.export.shipment'), ('res_id', '=', sid),
        ('body', 'like', needle)])

def protected():
    def nm(model, rid):
        r = E[model].browse(rid)
        return r.name if r.exists() else 'MISSING'

    def st(model, rid):
        r = E[model].browse(rid)
        if not r.exists():
            return 'MISSING'
        return r.state if 'state' in r._fields else nm(model, rid)

    def lot_split(lot_id):
        lot = E['stock.lot'].browse(lot_id)
        if not lot.exists():
            return 'MISSING'
        qs = E['stock.quant'].search([('lot_id', '=', lot_id)])
        return {'name': lot.name,
                'internal': round(sum(q.quantity for q in qs if q.location_id.usage == 'internal'), 4),
                'customer': round(sum(q.quantity for q in qs if q.location_id.usage == 'customer'), 4)}

    pr = {}
    pr['hist_ship_1537'] = (nm('lumber.export.shipment', 1537), st('lumber.export.shipment', 1537))
    pr['hist_cont_1509'] = (nm('lumber.container', 1509), st('lumber.container', 1509))
    pr['hist_pick_7065'] = (nm('stock.picking', 7065), st('stock.picking', 7065))
    pr['hist_line_1880'] = (E['lumber.shipment.line'].browse(1880).exists() and
                            E['lumber.shipment.line'].browse(1880).rebaja_state or 'MISSING')
    pr['hist_line_1881'] = (E['lumber.shipment.line'].browse(1881).exists() and
                            E['lumber.shipment.line'].browse(1881).rebaja_state or 'MISSING')
    pr['hist_lot_15970'] = lot_split(15970)
    pr['hist_lot_15971'] = lot_split(15971)
    pr['sv_ship_1528'] = (nm('lumber.export.shipment', 1528), st('lumber.export.shipment', 1528))
    pr['sv_cont_1500'] = (nm('lumber.container', 1500), st('lumber.container', 1500))
    pr['sv_pick_7062'] = (nm('stock.picking', 7062), st('stock.picking', 7062))
    pr['sv_op_5822'] = nm('madenat.sv.operation', 5822)
    pr['sv_imp_492'] = nm('madenat.sv.imputation', 492)
    pr['sv_guia_32298'] = nm('madenat.guia.processing', 32298)
    pr['sv_guia_32300'] = nm('madenat.guia.processing', 32300)
    pr['sv_lot_15892'] = lot_split(15892)
    pr['sv_lot_15894'] = lot_split(15894)
    pr['roll_vessel_619'] = nm('shipping.vessel', 619)
    pr['roll_vessel_620'] = nm('shipping.vessel', 620)
    pr['roll_voyage_565'] = nm('shipping.voyage', 565)
    for i in (1644, 1645, 1646):
        pr['roll_ship_%d' % i] = (nm('lumber.export.shipment', i), st('lumber.export.shipment', i))
    for i in (1629, 1630, 1631):
        pr['roll_cont_%d' % i] = (nm('lumber.container', i), st('lumber.container', i))
    for i in range(19699, 19705):
        pr['roll_audit_%d' % i] = (E['madenat.audit.log'].browse(i).exists() and
                                   E['madenat.audit.log'].browse(i).action_type or 'MISSING')
    for i in (1704, 1705, 1706, 1707):
        pr['draft_ship_%d' % i] = (nm('lumber.export.shipment', i), st('lumber.export.shipment', i))
    for i in (1678, 1679, 1680):

def attempt(label, fn):
    try:
        r = fn()
        extra = ''
        if isinstance(r, dict):
            extra = ' -> ' + str(r.get('res_model') or r.get('type'))
        P(label, 'NO_ERROR', extra)
    except Exception as e:  # noqa: BLE001
